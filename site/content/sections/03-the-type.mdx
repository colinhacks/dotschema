## The type

The type admits every value the schema allows. A keyword no TypeScript type can express, such as `minLength` or `format`, leaves the type as it is, and the validator checks it.

The type is `.Type` on the import:

```ts twoslash title="main.ts"
// @errors: 2322
import user from "./user.schema.jsonc";

const ada: user.Type = { name: "Ada", role: "admin" };
const bob: user.Type = { name: "Bob", role: "owner" };
```

A `/** */` comment above a property is its doc comment, which an editor shows on hover:

```jsonc title="user.schema.jsonc"
{
  "type": "object",
  "properties": {
    /** The name the user signs in with. */
    "name": { "type": "string", "minLength": 1 },
    "role": { "enum": ["admin", "member"] }
  },
  "required": ["name"],
  "additionalProperties": false
}
```

```ts twoslash title="main.ts"
declare const input: unknown;
// ---cut---
import user from "./user.schema.jsonc";

if (user.is(input)) {
  input.name;
  //    ^?
}
```

An `enum` or a `const` is a literal type:

```jsonc title="post.schema.jsonc"
{
  "type": "object",
  "properties": {
    "status": { "enum": ["draft", "published"] },
    "version": { "const": 1 }
  },
  "required": ["status", "version"],
  "additionalProperties": false
}
```

```ts twoslash title="main.ts"
declare const input: unknown;
// ---cut---
import post from "./post.schema.jsonc";

if (post.is(input)) {
  input;
//^?
}
```

A schema in `additionalProperties` types every other key:

```jsonc title="scores.schema.jsonc"
{
  "type": "object",
  "additionalProperties": { "type": "number" }
}
```

```ts twoslash title="main.ts"
declare const input: unknown;
// ---cut---
import scores from "./scores.schema.jsonc";

if (scores.is(input)) {
  input;
//^?
}
```

A `prefixItems` list is a tuple, its items required up to `minItems`:

```jsonc title="point.schema.jsonc"
{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "type": "array",
  "prefixItems": [
    { "type": "number" },
    { "type": "number" }
  ],
  "minItems": 2,
  "items": false
}
```

```ts twoslash title="main.ts"
import point from "./point.schema.jsonc";

type Point = point.Type;
//   ^?
```

A `oneOf` whose variants a `const` tells apart is a union that narrows:

```jsonc title="event.schema.jsonc"
{
  "type": "object",
  "oneOf": [
    {
      "properties": {
        "kind": { "const": "click" },
        "x": { "type": "number" }
      },
      "required": ["kind", "x"]
    },
    {
      "properties": {
        "kind": { "const": "visit" },
        "url": { "type": "string" }
      },
      "required": ["kind", "url"]
    }
  ]
}
```

```ts twoslash title="main.ts"
declare const input: unknown;
// ---cut---
import event from "./event.schema.jsonc";

if (event.is(input)) {
  if (input.kind === "click") {
    input.x;
    //    ^?
  } else {
    input.url;
    //    ^?
  }
}
```

An `if` that tests a property's listed values narrows the same way:

```jsonc title="payment.schema.jsonc"
{
  "type": "object",
  "properties": { "method": { "enum": ["card", "bank"] } },
  "required": ["method"],
  "if": { "properties": { "method": { "const": "card" } } },
  "then": {
    "properties": { "last4": { "type": "string" } },
    "required": ["last4"]
  },
  "else": {
    "properties": { "iban": { "type": "string" } },
    "required": ["iban"]
  }
}
```

```ts twoslash title="main.ts"
declare const input: unknown;
// ---cut---
import payment from "./payment.schema.jsonc";

if (payment.is(input) && input.method === "card") {
  input.last4;
  //    ^?
}
```
